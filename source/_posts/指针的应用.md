# 电梯等待时间

#### 题目：输入电梯停靠楼层的列表，以0结束输入，计算电梯到达最后楼层需要花费的时间。假设电梯每上一层需要花3秒时间，每下一层需要花2秒时间，电梯每停一次需要用时5秒

```c
#include <stdio.h>
int main(void)
{
    int i, number, n = 0, tm;
    int *p, ft[100];
    printf("Please input floors(Ends wit 0):\n");
    scanf("%d", &number);
    while (number != 0)
    {
        ft[n] = number;
        n++;
        scanf("%d", &number);
    }
    tm = 0;
    p = ft;
    for (i = 0; i < n - 1; i++)
    {
        if (*(p + 1) > *p)
            tm = tm + (*(p + 1) - *p) * 3 + 5;
        else
            tm = tm + (*p - *(p + 1)) * 2 + 5;
        p++;
    }
    printf("The time is: %d\n", tm);
    return 0;
}
```

# 冒泡排序

#### 题目：使用指针编写对整型数列进行冒泡法排序（升序）的函数

```c
#include <stdio.h>
void sort(int *list, int count);
int main(void)
{
    int a[6] = {2, 7, 4, 5, 6, 1}, i;
    sort(a, 6);
    printf("The result is: \n");
    for (i = 0; i < 6; i++)
        printf("%d ", a[i]);
    printf("\n");
    return 0;
}
void sort(int *list, int count)
{
    int i, j, k;
    for (i = 0; i < count - 1; i++)
    {
        for (j = 0; j < count - 1 - i; j++)
        {
            if (*(list + j) > *(list + j + 1))
            {
                k = *(list + j);
                *(list + j) = *(list + j + 1);
                *(list + j + 1) = k;
            }
        }
    }
}
```

# 简单加密器

#### 题目：最简单的加密器是将英文字符按照一定的规则进行转换，这里设置的规则为将字符替换成ASCII码表中该字符后面的第二个字符。比如'a'替换为'c'，'C'替换为'E'，'z'替换为'\\'。'Panda'替换为'Rcpfc'。

```c
#include <stdio.h>
#include <string.h>
int main(void)
{
    char str[100];
    char *p;
    printf("Input a string: \n");
    gets(str);
    p = str;
    while (*p != '\0')
    {
        *p += 2;
        p++;
    }
    printf("After being encrypted: %s\n", str);
    return 0;
}
```

# 查找和替换

#### 题目：在源字符串中查找目标 字符串，并进行替换。如输入源字符串"I am a boy!"，查找目标字符串"boy"，并将其替换为字符串"girl"，输出替换后的字符串"I am a girl!"，查找和替换功能要求使用函数实现。

```c
#include <stdio.h>
#include <string.h>
char *replace(char *s, char *ch, char *rh);
int main(void)
{
    char ch[40], rh[40], str[80], *p = NULL;
    printf("Please input the string:\n");
    gets(str);
    printf("Please input the string(search):\n");
    gets(ch);
    printf("Please input the string(replacce):\n");
    gets(rh);
    if ((p = replace(str, ch, rh)) == NULL)
    {
        printf("Not found!\n");
    }
    else
        printf("%s\n", p);
    return 0;
}
char *replace(char *s, char *ch, char *rh)
{
    int flag = 0, i;
    int m = strlen(ch), n = strlen(rh);
    char *p = s, rear[80];
    while (*p)
    {
        i = 0;
        while (*(p + i) != 0 && *(ch + i) != 0 && *(p + i) == *(ch + i))
            i++;
        if (*(ch + i) == 0)
        {
            strcpy(rear, p + m); // 将源字符串中查找到的目标字符串后面的所有字符串保存到rear中
            strcpy(p, rh);       // 将替换字符串复制到从p开始的位置
            strcpy(p + n, rear); // 将rear字符数组中的字符串复制到替换字符串后面
            flag = 1;
        }
        p++;
    }
    if (flag)
        return s;
    else
        return NULL;
}
```

# 分类统计

#### 题目：输入一串字符，统计各字母出现的次数（不区分大小写字母），并对出现的字母显示其次数和总字母书

```c
#include <stdio.h>
#include <ctype.h>
int main(void)
{
    char s[256];
    char *p;
    int a[26] = {0};
    int sumc = 0, i;
    printf("Enter a string:\n");
    gets(s);
    p = s;
    while (*p != '\0')
    {
        if (isupper(*p))
            a[*p - 'A']++;
        else if (islower(*p))
            a[*p - 'a']++;
        p++;
    }
    printf("The result is:\n");
    for (i = 0; i < 26; i++)
    {
        if (a[i])
        {
            sumc += a[i];
            printf("%c=%d\n", i + 'A', a[i]);
        }
    }
    printf("sum=%d\n", sumc);
    return 0;
}
```

# 字符串排序

#### 按照字母顺序对字符串排序

```c
#include <stdio.h>
#include <string.h>
int main(void)
{
    char *p[] = {"Juliet", "Romeo", "Hamlet", "Othello", "Macbeth"};
    char *t;
    int i, j;
    for (i = 0; i < 4; i++) // 选择排序法
    {
        for (j = i + 1; j < 5; j++)
        {
            if (strcmp(p[i], p[j]) > 0)
            {
                t = p[i];
                p[i] = p[j];
                p[j] = t;
            }
        }
    }
    for (i = 0; i < 5; i++)
        printf("%s\n", p[i]);
    return 0;
}
```
